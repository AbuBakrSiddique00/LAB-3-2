// Use 2901 Router 
// Use 2960 Switch

// PC configuration
1. pc1 - connect to router 1
IP address:      192.168.0.2
Subnet mask:     255.255.255.0
Default gateway: 192.168.0.1

2. pc2 - connect to router 1
IP address:      192.168.0.3
Subnet mask:     255.255.255.0
Default gateway: 192.168.0.1

3. pc3 - connect to router 2
IP address:      192.168.5.2
Subnet mask:     255.255.255.0
Default gateway: 192.168.5.1

4. pc4 - connect to router 2
IP address:      192.168.5.3
Subnet mask:     255.255.255.0
Default gateway: 192.168.5.1 


// Router configuration

Router 1: 
enable
configure terminal

interface g0/1
 ip address 10.0.0.2 255.0.0.0
 no shutdown
exit

interface g0/0
 ip address 192.168.0.1 255.255.255.0
 no shutdown
 exit
end


Router 2: 
enable
configure terminal

interface g0/1
 ip address 9.0.0.2 255.0.0.0
 no shutdown
exit

interface g0/0
 ip address 192.168.5.1 255.255.255.0
 no shutdown
 exit
end

Router 3:
enable
configure terminal

interface g0/0
 ip address 10.0.0.1 255.0.0.0
 no shutdown
 exit

interface g0/1
 ip address 9.0.0.1 255.0.0.0
 no shutdown
 exit
end

// Check configure in brief 
	show ip interface brief



// Static Route Configuration

Route 1: 
enable
configure terminal
ip route 192.168.5.0 255.255.255.0 10.0.0.1
end

Router 2:
enable
configure terminal
ip route 192.168.0.0 255.255.255.0 9.0.0.1
end

Router 3:
enable
configure terminal
ip route 192.168.0.0 255.255.255.0 10.0.0.2
ip route 192.168.5.0 255.255.255.0 9.0.0.2
end

// Now the most important part: Dynamic Routing
// First copy-paste the network diagram with static configuration

// Reconfigure the routers to RIP 

Router 1:
enable
configure terminal
router rip
 version 2
 no auto-summary
 network 10.0.0.0
 network 192.168.0.0
end

Router 2: 
enable
configure terminal
router rip
 version 2
 no auto-summary
 network 9.0.0.0
 network 192.168.5.0
end

Router 3: 
enable
configure terminal
router rip
 version 2
 no auto-summary
 network 10.0.0.0
 network 9.0.0.0
